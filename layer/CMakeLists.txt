cmake_minimum_required(VERSION 3.16)
project(RealityHatchLayer VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OPENXR_GENERATED_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../build/hello-xr-vulkan/include" CACHE PATH "Generated Khronos headers")
if(NOT EXISTS "${OPENXR_GENERATED_INCLUDE}/openxr/openxr_loader_negotiation.h")
    message(FATAL_ERROR "Build hello_xr first to generate the Khronos headers.")
endif()
add_library(RealityHatchLayer SHARED realityhatch.cpp)
target_include_directories(RealityHatchLayer PRIVATE "${OPENXR_GENERATED_INCLUDE}")
target_compile_definitions(RealityHatchLayer PRIVATE XR_NO_PROTOTYPES WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(RealityHatchLayer PRIVATE /W4 /permissive- /EHsc)
set_target_properties(RealityHatchLayer PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../bin/layers")
configure_file(RealityHatch.json "${CMAKE_CURRENT_SOURCE_DIR}/../bin/layers/RealityHatch.json" COPYONLY)
add_executable(RealityHatchDispatchTests realityhatch.cpp tests/dispatch.cpp)
target_include_directories(RealityHatchDispatchTests PRIVATE "${OPENXR_GENERATED_INCLUDE}")
target_compile_definitions(RealityHatchDispatchTests PRIVATE XR_NO_PROTOTYPES WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(RealityHatchDispatchTests PRIVATE /W4 /permissive- /EHsc)
enable_testing()
add_test(NAME dispatch-forwarding COMMAND RealityHatchDispatchTests)
